CHANNEL_ID = "1410924658840178738"

# Aapka trading psychology wala paragraph
my_paragraph = "Trading psychology is the blend of emotions, mental states, and behavioral biases that steer how a person makes decisions in financial markets."

def send_message():
    url = f"https://discord.com/api/v9/channels/{CHANNEL_ID}/messages"
    
    headers = {
        "Authorization": TOKEN,
        "Content-Type": "application/json"
    }
    
    payload = {
        "content": my_paragraph
    }
    
    try:
        response = requests.post(url, json=payload, headers=headers)
        if response.status_code == 200:
            print("Message successfully bhej diya gaya!")
        else:
            print(f"Error aaya: {response.status_code} - {response.text}")
    except Exception as e:
        print("Exception:", e)

# Loop jo har 30 second baad message bhejega
while True:
    send_message()
    time.sleep(30)
